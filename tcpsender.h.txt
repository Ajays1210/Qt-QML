#ifndef TCPSENDER_H
#define TCPSENDER_H

#include <QObject>
#include <QTcpSocket>

class TcpSender : public QObject
{
    Q_OBJECT

public:
    explicit TcpSender(QObject *parent = nullptr);

public slots:
    void start();
    void stop();

    void sendAdd(int uniqueID, double lat,
                 double longitude,
                 const QString &comment);

    void sendUpdate(int uniqueID, double lat,
                    double longitude,
                    const QString &comment);

    void sendDelete(int uniqueID);

private:
    bool ensureConnected(); // Connects to the server if needed.

private:
    QTcpSocket *m_socket;
};

#endif
