#ifndef UDPRECEIVER_H
#define UDPRECEIVER_H

#include <QObject>
#include <QUdpSocket>

class UdpReceiver : public QObject
{
    Q_OBJECT

public:
    explicit UdpReceiver(QObject *parent = nullptr);

public slots:
    void start(); // Bind the UDP socket and start receiving.
    void stop();  // Close the UDP socket.

signals:
    void addResponseReceived(quint8 respID, quint8 ack,
                             quint32 uniqueID, float lat,
                             float longitude,
                             const QString &comment);

    void deleteResponseReceived(quint8 respID, quint8 ack,
                                quint32 uniqueID);

private slots:
    void readPendingDatagrams(); // Reads all available UDP packets.

private:
    QUdpSocket *m_socket = nullptr;
};

#endif
